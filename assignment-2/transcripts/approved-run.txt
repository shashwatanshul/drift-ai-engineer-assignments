```python
def merge_intervals(intervals):
    """
    Merge a list of intervals.

    Parameters
    ----------
    intervals : list of [int, int] or [(int, int), ...]
        Each interval is a pair [start, end] where start <= end.

    Returns
    -------
    list of [int, int]
        A new list of non‑overlapping intervals covering the same points,
        sorted by their start value. Intervals that overlap or touch are merged.
    """
    if not intervals:
        return []

    # Sort intervals by their start value (and by end to stabilize equal starts)
    sorted_intervals = sorted(intervals, key=lambda x: (x[0], x[1]))

    merged = []
    cur_start, cur_end = sorted_intervals[0]

    for start, end in sorted_intervals[1:]:
        # If the current interval overlaps or touches the previous one, merge it
        if start <= cur_end:          # overlap or touch
            cur_end = max(cur_end, end)
        else:
            merged.append([cur_start, cur_end])
            cur_start, cur_end = start, end

    # Append the last accumulated interval
    merged.append([cur_start, cur_end])
    return merged
```
