from typing import List

def merge_intervals(intervals: List[List[int]]) -> List[List[int]]:
    """
    Merge overlapping or touching intervals from the input list.

    Args:
        intervals (List[List[int]]): A list of intervals, each represented as [start, end].

    Returns:
        List[List[int]]: A new list of merged, non-overlapping intervals sorted by start.

    Notes:
        - Intervals that overlap or touch (e.g., [1, 3] and [3, 5]) are merged.
        - The input list is not modified.
        - Assumes intervals are well-formed with start <= end.
    """
    if not intervals:
        return []

    # Make a shallow copy to avoid modifying the caller's list
    intervals_copy = [interval[:] for interval in intervals]

    # Sort intervals by start time
    intervals_copy.sort(key=lambda x: x[0])

    merged = []
    current_start, current_end = intervals_copy[0]

    for start, end in intervals_copy[1:]:
        # If intervals overlap or touch, merge them
        if start <= current_end:
            current_end = max(current_end, end)
        else:
            # No overlap, push the current interval and reset
            merged.append([current_start, current_end])
            current_start, current_end = start, end

    # Append the last interval
    merged.append([current_start, current_end])

    return merged
