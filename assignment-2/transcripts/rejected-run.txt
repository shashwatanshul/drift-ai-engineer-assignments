def merge_intervals(intervals):
    intervals = [i[:] for i in intervals]
    intervals.sort()
    while True:
        merged = False
        for i in range(len(intervals)-1):
            if intervals[i][1] >= intervals[i+1][0]:
                intervals[i][1] = max(intervals[i][1], intervals[i+1][1])
                intervals.pop(i+1)
                merged = True
                break
        if not merged:
            break
    return intervals
